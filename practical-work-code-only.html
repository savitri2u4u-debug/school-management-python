# Program 17
def factorial(n):
    if n == 0 or n == 1:
        return 1
    return n * factorial(n - 1)
def fib(n):
    if n <= 1:
        return n
    return fib(n - 1) + fib(n - 2)
num = int(input("Enter a number: "))
print("Factorial of", num, "=", factorial(num))
print("Fibonacci series:", end=" ")
for i in range(num):
    print(fib(i), end=" ")


# Program 18
def binary_search(lst, key):
    low, high = 0, len(lst) - 1
    while low <= high:
        mid = (low + high) // 2
        if lst[mid] == key:
            return mid
        elif lst[mid] < key:
            low = mid + 1
        else:
            high = mid - 1
    return -1
data = [5, 12, 23, 34, 45, 56, 67, 78, 89]
key = int(input("Enter element to search: "))
pos = binary_search(data, key)
if pos != -1:
    print(key, "found at index", pos)
else:
    print(key, "not found in the list")

        else:

            print("Stack is empty")
        else:
            print("Top element:", stack[-1])
    elif ch == 4:
        if len(stack) == 0:
            print("Stack is empty")
        else:
            print("Stack (top to bottom):", stack[::-1])
    elif ch == 5:
        print("Program ended")
        break
    else:
        print("Invalid choice")


# Program 20
def is_balanced(expr):
    stack = []
    pairs = {')': '(', ']': '[', '}': '{'}
    for ch in expr:
        if ch in "([{":
            stack.append(ch)
        elif ch in ")]}":
            if not stack or stack.pop() != pairs[ch]:
                return False
    return len(stack) == 0
for e in ["(a+b)*{c-d}", "((a+b)*[c-d)]", "{[(x+y)*z]}"]:
    if is_balanced(e):
        print(e, "->", "Balanced")
    else:
        print(e, "->", "Not Balanced")
