# Q1. Import Modules and Load Data from File
import pickle
import os

students = {}
teachers = {}
books = {}

def load_data():
    global students, teachers, books
    if os.path.exists("school.dat"):
        try:
            with open("school.dat", "rb") as f:
                students, teachers, books = pickle.load(f)
        except (EOFError, pickle.PickleError, ValueError):
            print("Saved data file could not be read. Starting with empty records.")

load_data()


# Q2. Save Data to Binary File
def save_data():
    with open("school.dat", "wb") as f:
        pickle.dump((students, teachers, books), f)

print("Data Storage Helper Ready!")

# Q4. Add New Student Admission
def add_student():
    print("\n===== NEW STUDENT ADMISSION =====")
    adm_no = str(max([int(x) for x in students], default=100) + 1)
    name = input("Enter Student Name: ")
    class_sec = input("Enter Class and Section: ")
    phone = input("Enter Parent Phone Number: ")
    students[adm_no] = {
        "name": name,
        "class": class_sec,
        "phone": phone,
        "attendance": "Not Marked",
        "marks": 0,
        "fee_status": "Pending"
    }
    save_data()
    print("Student Admitted! Admission No:", adm_no)

# Q12. Attendance Summary
def attendance_summary():
    print("\n===== ATTENDANCE SUMMARY =====")
    present = 0
    absent = 0
    not_marked = 0
    for d in students.values():
        if d["attendance"] == "Present":
            present += 1
        elif d["attendance"] == "Absent":
            absent += 1
        else:
            not_marked += 1
    print("Present    :", present)
    print("Absent     :", absent)
    print("Not Marked :", not_marked)


# Q13. Enter Student Exam Marks
def enter_marks():
    print("\n===== ENTER STUDENT EXAM MARKS =====")
    adm_no = input("Enter Admission Number: ")
    if adm_no in students:
        eng = float(input("English Marks: "))
        phy = float(input("Physics Marks: "))
        chem = float(input("Chemistry Marks: "))
        math = float(input("Maths Marks: "))
        cs = float(input("Computer Science Marks: "))
        total = eng + phy + chem + math + cs
        students[adm_no]["marks"] = total
        save_data()
        print("Total Marks Saved:", total, "/ 500")
    else:
        print("Student Not Found!")

# Q15. School Topper
def school_topper():
    print("\n===== SCHOOL TOPPER =====")
    if len(students) == 0:
        print("No Student Records Found!")
    else:
        top = ""
        best = -1
        for adm_no, d in students.items():
            if d["marks"] > best:
                best = d["marks"]
                top = adm_no
        print("Topper    :", students[top]["name"])
        print("Adm No.   :", top)
        print("Class     :", students[top]["class"])
        print("Marks     :", best, "/ 500")

# Q26. Search Book by Title
def search_book():
    print("\n===== SEARCH BOOK BY TITLE =====")
    word = input("Enter Title / Keyword: ").lower()
    found = False
    for book_id, info in books.items():
        if word in info["title"].lower():
            print(book_id, "|", info["title"], "|", info["author"], "|",
                  info["copies"])
            found = True
    if not found:
        print("No Matching Book Found!")


# Q27. Issue Library Book
def issue_book():
    print("\n===== ISSUE LIBRARY BOOK =====")
    book_id = input("Enter Book ID: ")
    if book_id in books:
        if books[book_id]["copies"] > 0:
            books[book_id]["copies"] -= 1
            save_data()
            print("Book Issued! Remaining Copies:",
                  books[book_id]["copies"])
        else:
            print("Book Out of Stock!")
    else:
        print("Book ID Not Found!")


# Q28. Return Library Book
def return_book():
    print("\n===== RETURN LIBRARY BOOK =====")
    book_id = input("Enter Book ID: ")
    if book_id in books:
        books[book_id]["copies"] += 1
        save_data()
        print("Book Returned! Available Copies:", books[book_id]["copies"])
    else:
        print("Book ID Not Found!")

    }
    while True:
        print("\n1-5:Student 6-7:Attend 8-9:Marks")
        print("10-11:Fee 12-15:Staff 16-20:Book")
        print("21-26:Extra Reports 27:Exit")
        ch = input("Enter Choice: ")
        if ch == "27":
            break
